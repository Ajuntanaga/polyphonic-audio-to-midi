cmake_minimum_required(VERSION 3.20)
project(m3_experimental_neural_worker LANGUAGES CXX)
# Deliberately standalone: neither the production plugin nor its SDK depends on ORT.
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(Threads REQUIRED)
if(NOT CMAKE_SYSTEM_NAME STREQUAL "Linux")
  message(FATAL_ERROR "This bounded measurement harness currently targets Linux only")
endif()
add_library(callback_allocations STATIC allocation_guard.cpp)
target_link_options(callback_allocations INTERFACE
  -Wl,--wrap=malloc -Wl,--wrap=calloc -Wl,--wrap=realloc
  -Wl,--wrap=aligned_alloc -Wl,--wrap=posix_memalign -Wl,--wrap=free)
add_executable(neural_core_tests stream_test.cpp)
target_link_libraries(neural_core_tests PRIVATE callback_allocations Threads::Threads)
enable_testing()
add_test(NAME neural_core COMMAND neural_core_tests)
if(NOT ORT_ROOT)
  message(FATAL_ERROR "Pass the explicit ONNX Runtime 1.20.1 CPU SDK directory as ORT_ROOT")
endif()
find_path(ORT_INCLUDE onnxruntime_cxx_api.h PATHS "${ORT_ROOT}/include" NO_DEFAULT_PATH REQUIRED)
find_library(ORT_LIBRARY onnxruntime PATHS "${ORT_ROOT}/lib" NO_DEFAULT_PATH REQUIRED)
add_executable(neural_ort_harness ort_harness.cpp)
target_include_directories(neural_ort_harness PRIVATE "${ORT_INCLUDE}")
target_link_libraries(neural_ort_harness PRIVATE "${ORT_LIBRARY}" callback_allocations Threads::Threads)
foreach(target callback_allocations neural_core_tests neural_ort_harness)
  target_compile_options(${target} PRIVATE -Wall -Wextra -Wpedantic -ffp-contract=off)
endforeach()
